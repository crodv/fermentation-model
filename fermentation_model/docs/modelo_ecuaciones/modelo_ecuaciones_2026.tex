\begin{center}
{\Large\bfseries Modelo de fermentación y capa CO\textsubscript{2} --- ecuaciones y parámetros}\\[2pt]
Estado vigente (mosto natural; capa CO\textsubscript{2} canónica SCCM\_CORRECTED). 2026-09-19.
\end{center}

Estados: $X,X_d$ biomasa viable/muerta, $N$ nitrógeno asimilable, $G,F$ glucosa/fructosa,
$E$ etanol, $Gly$ glicerol $[\kgm]$. $T$ temperatura $[\mathrm{K}]$ (input medido), $t$ $[\mathrm{h}]$.

\section*{1. Modelo upstream (ODE central)}

Pulsos (canales $N,G,F,E,X$): salto exacto del estado en $t_p$ entre tramos de integración.
\begin{align}
\dot X &= (\mu - k_d)\,X \tag{U1}\\
\dot X_d &= k_d\,X \tag{U2}\\
\dot N &= -\,q_N\,a_\mu\,\phi_n\,X \tag{U3}\\
\dot G &= -\Big(q_{XG}\,a_\mu\phi_n + q_{EG}\,f_G + m\,\frac{G}{G+F}\Big)X \tag{U4}\\
\dot F &= -\Big(q_{XF}\,a_\mu\phi_n + q_{EF}\,f_F + m\,\frac{F}{G+F}\Big)X \tag{U5}\\
\dot E &= (\beta_G+\beta_F)\,X \tag{U6}\\
\dot{Gly} &= (\gamma_{G0}\,f_G + \gamma_{F0}\,f_F)\,X \tag{U7}
\end{align}

\subsection*{Términos cinéticos}
\begin{align}
a_\mu &= \exp\!\Big[\tfrac{E_{ac}(T-300)}{300\,R\,T}\Big], \qquad
a_\beta = \exp\!\Big[\tfrac{E_{afe}(T-296.15)}{296.15\,R\,T}\Big], \qquad
a_k = \exp\!\Big[\tfrac{E_{ak}(T-293.15)}{293.15\,R\,T}\Big] \tag{U8}\\
k_n &= \frac{\mu_0}{s_N}a_k, \qquad
k_G = \frac{\beta_{G0}}{s_G}a_k, \qquad
k_F = \frac{\beta_{F0}}{s_F}a_k, \qquad
\imath_e = \frac{\imath_E}{a_k}, \qquad
\imath_g = \frac{\imath_G}{a_k} \tag{U9}\\
\phi_n &= \frac{N}{N+k_n}, \qquad
\phi_G = \frac{G}{G+k_G}, \qquad
\phi_F = \frac{F}{F+k_F} \tag{U10}\\
\Phi_E &= \frac{1}{1+\imath_e\,E}, \qquad
\Phi_{G\to F} = \frac{1}{1+\imath_g\,G} \tag{U11}\\
\mu &= \mu_0\,a_\mu\,\phi_n \tag{U12}\\
f_G &\equiv a_\beta\,\phi_G\,\Phi_E, \qquad
f_F \equiv a_\beta\,\phi_F\,\Phi_{G\to F}\,\Phi_E, \qquad
\beta_G = \beta_{G0}f_G, \qquad
\beta_F = \beta_{F0}f_F \tag{U13}\\
m &= m_0\exp\!\Big[\tfrac{E_{am}(T-293.3)}{293.3\,R\,T}\Big]\cdot\frac{G+F}{G+F+c_m} \tag{U14}\\
k_d &= \begin{cases}
K_{d0}\exp\!\Big[C_{de}E+\tfrac{E_{td}(T-305.65)}{305.65\,R\,T}\Big] & T\ge T_D(E)\\
0 & T<T_D(E)
\end{cases} \tag{U15}\\
T_D(E) &= -0.0001\,E^3+0.0049\,E^2-0.1279\,E+315.89 \tag{U16}
\end{align}

\section*{2. Capa CO\textsubscript{2}}

\begin{align}
q_{bio} &= \chi_{CO_2/E}\,\max(\beta_G+\beta_F,0)\,X, \qquad
\chi_{CO_2/E}=\frac{44.01}{2(46.07)}=0.4776 \tag{C1}
\end{align}

\textbf{Gate de activación química (bracket offline).} $t_U$: primera muestra química con
$\Delta(G+F)\ge 5$ \gL~o $\Delta E\ge 2$ \gL; $t_L$: muestra anterior.
\begin{align}
t_s &= t_L + f_s\,(t_U-t_L), \qquad
\Delta t_a = f_d\,(t_U-t_L), \qquad
z = \operatorname{clip}\!\Big(\frac{t-t_s}{\Delta t_a},0,1\Big) \tag{C2}\\
a_{chem}(t) &= 3z^2-2z^3 \tag{C3}
\end{align}

\textbf{O\textsubscript{2} (pool interno, Euler; no es estado del central).}
$O_2(0)=s_{O_2}\,O_{2,sat}(T,E,G,F)$.
\begin{align}
u &= q_{O_2,max}\,X\,\frac{O_2}{K_{O_2}+O_2}, \qquad
O_2 \leftarrow \max\!\big(O_2-\Delta t\,u,\,0\big) \tag{C4}\\
\phi_{ana} &= \frac{K_{ana}^{h}}{K_{ana}^{h}+O_2^{h}}, \qquad
f_{ferm} = 0.08+0.92\,\phi_{ana} \tag{C5}\\
q_{resp} &= \frac{44.01}{32000}\,u \tag{C6}\\
q_{prod} &= a_{chem}\,\big(q_{bio}\,f_{ferm} + q_{resp}\big) \tag{C7}
\end{align}

\textbf{Pulso de N (salto + transición de utilización).} Salto exacto
$N(t_p^+)=N(t_p^-)+\Delta N$ en el central; en la capa, con
$r(t)=\operatorname{clip}\!\big(\frac{t-t_p}{t_{rise}},0,1\big)$ y contrafactual sin pulso:
\begin{align}
X_{eff} &= \max\!\big(X^{sin} + r\,\Delta X,\,0\big) \tag{C8}\\
g_{act} &= 1+(g_{pulse}-1)\,r \tag{C9}\\
q_{bio}^{pulso} &= \max\!\big(g_{act}\,q_{bio}^{sin} + r\,\Delta q_{bio},\,0\big) \tag{C10}
\end{align}

\textbf{CO\textsubscript{2} disuelto (pool, Euler) y liberación continua.}
\begin{align}
C^*_{sat} &= 1.69\,e^{-0.032\,(T-20)}\,e^{0.0016\,E}\,e^{-0.0012\,(G+F)}, \qquad
C_{sat} = s_{sat}\,C^*_{sat} \tag{C11}\\
f_{rel} &= 0.05+0.95\,\frac{C}{C+C_{sat}}, \qquad
\hat q_{gas} = \min\!\big(k_C\,C\,f_{rel},\;\; C/\Delta t + q_{prod}\big) \tag{C12}\\
C &\leftarrow \max\!\big(C+\Delta t\,(q_{prod}-\hat q_{gas}),\,0\big) \tag{C13}
\end{align}

\textbf{Observación.} $\hat y = g_{m}\,\hat q_{gas}$; la señal observada SCCM se convierte con
\begin{align}
\kappa &= \frac{44.0095\cdot 60\cdot 0.74}{1000\cdot 24.16\cdot V} = 0.0404392 \quad (V=2~\mathrm{L})
\tag{C14}
\end{align}
